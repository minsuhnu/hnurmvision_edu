cmake_minimum_required(VERSION 3.16)
project(hnurmvision_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(hnurm_interfaces REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs highgui calib3d dnn video videoio)
find_package(Threads REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
add_library(training_io io/config.cpp)
target_include_directories(training_io PUBLIC include io ${OpenCV_INCLUDE_DIRS})
target_link_libraries(training_io Eigen3::Eigen yaml-cpp ${OpenCV_LIBS})
ament_target_dependencies(training_io rclcpp sensor_msgs hnurm_interfaces)
target_compile_options(training_io PRIVATE -Wall -Wextra -Wpedantic)
add_library(training_lessons lessons/lesson02_io.cpp
 lessons/lesson03_detection.cpp lessons/lesson04_tracking.cpp
 lessons/lesson05_aiming.cpp src/pose_inference.cpp src/lesson04_runtime.cpp)
target_link_libraries(training_lessons PUBLIC training_io)
add_executable(training src/main.cpp src/realtime_processor.cpp)
target_link_libraries(training PRIVATE training_lessons Threads::Threads)
add_executable(lesson04_demo src/lesson04_demo.cpp)
target_link_libraries(lesson04_demo PRIVATE training_lessons Threads::Threads)
target_compile_options(lesson04_demo PRIVATE -Wall -Wextra -Wpedantic)

# 第四讲离线测试，默认不构建。
option(BUILD_LESSON04_PNP_TEST "Build the isolated lesson 04 PnP core test" OFF)
if(BUILD_LESSON04_PNP_TEST)
  enable_testing()
  add_executable(lesson04_pnp_test test/lesson04_pnp_test.cpp lessons/lesson04_tracking.cpp src/lesson04_runtime.cpp)
  target_include_directories(lesson04_pnp_test PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(lesson04_pnp_test PRIVATE Eigen3::Eigen yaml-cpp ${OpenCV_LIBS})
  add_test(NAME lesson04_pnp_test COMMAND lesson04_pnp_test ${CMAKE_CURRENT_SOURCE_DIR}/configs/simulation.yaml)
  set_tests_properties(lesson04_pnp_test PROPERTIES TIMEOUT 45)
  add_executable(lesson04_tracking_test test/lesson04_tracking_test.cpp)
  target_link_libraries(lesson04_tracking_test PRIVATE training_lessons)
  target_compile_options(lesson04_tracking_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME lesson04_tracking_test COMMAND lesson04_tracking_test ${CMAKE_CURRENT_SOURCE_DIR}/configs/lesson04.yaml)
  set_tests_properties(lesson04_tracking_test PROPERTIES TIMEOUT 45)
endif()
